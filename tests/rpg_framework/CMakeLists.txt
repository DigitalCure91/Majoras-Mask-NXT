cmake_minimum_required(VERSION 3.26)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RpgRegressionTests LANGUAGES CXX)
    enable_testing()
endif()
get_filename_component(RPG_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(RpgCoreTests core_tests.cpp)
add_executable(RpgSaveTests save_tests.cpp "${RPG_SOURCE_ROOT}/mm/2s2h/SaveManager/Migrations/9.cpp")
target_compile_features(RpgCoreTests PRIVATE cxx_std_20)
target_compile_features(RpgSaveTests PRIVATE cxx_std_20)
target_include_directories(RpgSaveTests PRIVATE "${RPG_SOURCE_ROOT}/mm")
target_link_libraries(RpgSaveTests PRIVATE nlohmann_json::nlohmann_json)
add_test(NAME Rpg.Core COMMAND RpgCoreTests)
add_test(NAME Rpg.SaveMigration COMMAND RpgSaveTests)
